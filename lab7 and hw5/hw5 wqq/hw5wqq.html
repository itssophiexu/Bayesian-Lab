Question 1

The independent binomial likelihood is \[ L(\beta_0,\beta_1) = \prod_{j=1}^{8} \binom{n_j}{z_j} p_j^{z_j}(1-p_j)^{n_j-z_j}, \qquad p_j=\frac{e^{\beta_0+\beta_1x_j}} {1+e^{\beta_0+\beta_1x_j}}. \]

Up to an additive constant, the log likelihood is \[ \begin{aligned} \ell(\beta_0,\beta_1) &= \sum_{j=1}^{8} \left[z_j\log p_j+(n_j-z_j)\log(1-p_j)\right]\\ &= \sum_{j=1}^{8} \left[ z_j(\beta_0+\beta_1x_j) -n_j\log\left(1+e^{\beta_0+\beta_1x_j}\right) \right]. \end{aligned} \]

The independent normal priors give \[ \log p(\beta_0)+\log p(\beta_1) = -\frac{\beta_0^2}{2(2.5^2)} -\frac{\beta_1^2}{2(2.5^2)} +C. \]

Adding the log likelihood and log priors gives \[ \boxed{ \begin{aligned} \log\tilde p(\beta_0,\beta_1\mid z,n,x) ={}& \sum_{j=1}^{8} \left[ z_j(\beta_0+\beta_1x_j) -n_j\log\left(1+e^{\beta_0+\beta_1x_j}\right) \right]\\ &-\frac{\beta_0^2+\beta_1^2}{12.5}, \end{aligned} } \] up to an additive constant.

Question 2

For the symmetric normal random-walk proposal, the acceptance probability is \[ \alpha = \min\left\{ 1, \frac{\tilde p(\boldsymbol{\beta}^{\star}\mid z,n,x)} {\tilde p(\boldsymbol{\beta}^{(t)}\mid z,n,x)} \right\}. \]

Equivalently, \[ \log\alpha = \min\left\{ 0, \log\tilde p(\boldsymbol{\beta}^{\star}\mid z,n,x) - \log\tilde p(\boldsymbol{\beta}^{(t)}\mid z,n,x) \right\}. \]

The posterior normalizing constant cancels because it is the same at both states. The proposal-density terms cancel because the proposal is symmetric.

log_post <- function(beta, x, n, z) {
  p <- plogis(beta[1] + beta[2] * x)
  sum(dbinom(z, size = n, prob = p, log = TRUE)) +

}

Question 4

rw_metropolis <- function(x, n, z, iterations = 30000,
                          prop.sd = 0.04,
                          start = c(0, 0)) {
  beta.current <- start
  log.current <- log_post(beta.current, x, n, z)

  draws <- matrix(
    NA_real_,
    nrow = iterations,
    ncol = 2,
    dimnames = list(NULL, c("beta0", "beta1"))
  )

  accepted <- 0L

  for (t in seq_len(iterations)) {
    beta.proposed <- beta.current + rnorm(2, 0, prop.sd)
    log.proposed <- log_post(beta.proposed, x, n, z)
    log.alpha <- log.proposed - log.current

    if (log(runif(1)) < min(0, log.alpha)) {
      beta.current <- beta.proposed
      log.current <- log.proposed
      accepted <- accepted + 1L
    }

    draws[t, ] <- beta.current
  }

  list(
    draws = draws,
    acceptance_rate = accepted / iterations
  )
}

set.seed(402)

prop.sd <- 0.04

run1 <- rw_metropolis(
  x, n, z,
  iterations = 30000,
  prop.sd = prop.sd,
  start = c(0, 0)
)
